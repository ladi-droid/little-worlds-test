/* Little Worlds Google Sheet script.
   1. Feedback: receives answers from feedback.html and writes them to the "Answers" tab.
      One row per person. The row updates after each answer, so half-finished forms still show up.
      Columns appear in question order the first time someone answers.
   2. Usage: receives small anonymous events from the game and the builder (lw-ping.js) and adds one row each
      to the "Events" tab (no names or emails, just a random browser id, the game's link name and the device type).
   3. Admin page: admin.html reads both tabs back through doGet, only with the admin key.
      Set the key once in Apps Script: Project Settings > Script properties > add ADMIN_KEY = (your own password). */
const SHEET = 'Answers';
const EVENTS = 'Events';
const EVENT_HEAD = ['Time', 'Browser', 'Link', 'Event', 'Trip', 'Progress %', 'Seconds', 'Specials', 'Extra', 'Ladi test', 'Device', 'Home screen', 'Page'];

function doPost(e) {
  const d = JSON.parse(e.postData.contents);
  if (d.kind === 'event') return addEvent(d);
  const lock = LockService.getScriptLock();
  lock.waitLock(15000);
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sh = ss.getSheetByName(SHEET) || ss.insertSheet(SHEET);

    // Header row: fixed columns, then every question column in order.
    let head = sh.getLastColumn() ? sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0].map(String) : [];
    const want = ['Response id', 'Started', 'Last answer', 'Finished'].concat(d.columns || []);
    want.forEach(k => { if (head.indexOf(k) < 0) head.push(k); });
    sh.getRange(1, 1, 1, head.length).setValues([head]).setFontWeight('bold');
    sh.setFrozenRows(1);

    // Find this person's row, or add one.
    const n = sh.getLastRow();
    const ids = n > 1 ? sh.getRange(2, 1, n - 1, 1).getValues().map(r => String(r[0])) : [];
    let row = ids.indexOf(String(d.id));
    const now = new Date();
    if (row < 0) {
      row = n + 1;
      sh.getRange(row, 1, 1, 2).setValues([[String(d.id), now]]);
    } else {
      row += 2;
    }

    const set = (k, v) => sh.getRange(row, head.indexOf(k) + 1).setValue(v);
    set('Last answer', now);
    set('Finished', d.finished ? 'Yes' : 'Not yet');
    const a = d.answers || {};
    (d.columns || []).forEach(k => set(k, k in a ? a[k] : ''));

    return ContentService.createTextOutput('ok');
  } finally {
    lock.releaseLock();
  }
}

// one usage event = one new row on the Events tab
function addEvent(d) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sh = ss.getSheetByName(EVENTS);
  if (!sh) {
    sh = ss.insertSheet(EVENTS);
    sh.getRange(1, 1, 1, EVENT_HEAD.length).setValues([EVENT_HEAD]).setFontWeight('bold');
    sh.setFrozenRows(1);
  }
  const s = v => String(v == null ? '' : v).slice(0, 200);
  sh.appendRow([new Date(), s(d.dev), s(d.game), s(d.ev), s(d.trip), d.progress === '' ? '' : Number(d.progress),
    d.secs === '' ? '' : Number(d.secs), d.specials === '' ? '' : Number(d.specials), s(d.extra), d.admin ? 'yes' : '',
    s(d.device), d.home ? 'yes' : '', s(d.page)]);
  return ContentService.createTextOutput('ok');
}

// admin.html: ?key=...  returns both tabs as JSON. Without the right key it only says the script is running.
function doGet(e) {
  const key = PropertiesService.getScriptProperties().getProperty('ADMIN_KEY');
  const p = (e && e.parameter) || {};
  if (!key || p.key !== key) {
    const msg = p.key ? { error: key ? 'wrong key' : 'no ADMIN_KEY set in Script properties' } : null;
    return msg ? json(msg) : ContentService.createTextOutput('Little Worlds feedback is running.');
  }
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const tab = (name, limit) => {
    const sh = ss.getSheetByName(name);
    if (!sh || sh.getLastRow() < 1) return { head: [], rows: [] };
    const all = sh.getDataRange().getValues();
    const head = all.shift().map(String);
    const rows = all.slice(-limit).map(r => r.map(v => v instanceof Date ? v.toISOString() : v));
    return { head, rows };
  };
  return json({ answers: tab(SHEET, 5000), events: tab(EVENTS, 50000), at: new Date().toISOString() });
}

function json(o) {
  return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON);
}
